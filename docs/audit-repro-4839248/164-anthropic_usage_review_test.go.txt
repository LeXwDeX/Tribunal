type auditTransport struct {
	bodies []string
	calls  atomic.Int64
}

func (tr *auditTransport) RoundTrip(req *http.Request) (*http.Response, error) {
	if req.URL.Host != "audit.invalid" {
		return nil, fmt.Errorf("unexpected host %q", req.URL.Host)
	}
	call := int(tr.calls.Add(1))
	if call > len(tr.bodies) {
		return nil, errors.New("unexpected extra request")
	}
	return &http.Response{
		StatusCode: 200,
		Header:     http.Header{"Content-Type": []string{"application/json"}},
		Body:       io.NopCloser(strings.NewReader(tr.bodies[call-1])),
		Request:    req,
	}, nil
}

func auditClient(t *testing.T, protocol string, bodies ...string) (llm.Client, *auditTransport) {
	t.Helper()
	t.Setenv("AUDIT_ONLY_KEY", "fake-key")
	transport := &auditTransport{bodies: bodies}
	registry, err := llm.NewRegistry(map[string]config.ProviderConfig{
		"fake": {Protocol: protocol, BaseURL: "https://audit.invalid", Model: "fake", APIKeyEnv: "AUDIT_ONLY_KEY", MaxTokens: 128},
	}, llm.WithHTTPClient(&http.Client{Transport: transport}))
	if err != nil {
		t.Fatal(err)
	}
	client, err := registry.Client("fake")
	if err != nil {
		t.Fatal(err)
	}
	return client, transport
}

func auditRequest() *llm.Request {
	return &llm.Request{Messages: []llm.Message{{Role: llm.RoleUser, Parts: []llm.Part{{Type: llm.PartText, Text: "audit"}}}}}
}

type auditNormalizedClient struct{ llm.Client }

func (c auditNormalizedClient) Generate(ctx context.Context, req *llm.Request) (*llm.Response, error) {
	response, err := c.Client.Generate(ctx, req)
	if response != nil {
		copy := *response
		copy.Usage.InputTokens += copy.Usage.CacheReadTokens + copy.Usage.CacheWriteTokens
		return &copy, err
	}
	return response, err
}

func TestAuditAdapterComparison(t *testing.T) {
	for _, tc := range []struct {
		protocol string
		body     string
		input    int64
	}{
		{"anthropic", auditAnthropicFirst, 10},
		{"openai-chat", auditChat, 15},
		{"openai-responses", auditResponses, 15},
	} {
		t.Run(tc.protocol, func(t *testing.T) {
			client, transport := auditClient(t, tc.protocol, tc.body)
			resp, err := client.Generate(context.Background(), auditRequest())
			if err != nil {
				t.Fatal(err)
			}
			want := (llm.Usage{InputTokens: tc.input, OutputTokens: 5, CacheReadTokens: 2, CacheWriteTokens: 3, ReasoningTokens: 4})
			if resp.Usage != want || transport.calls.Load() != 1 {
				t.Fatalf("usage=%+v requests=%d, want %+v and 1 request", resp.Usage, transport.calls.Load(), want)
			}
			t.Logf("synthetic %s: usage=%+v, budget input+output=%d", tc.protocol, resp.Usage, resp.Usage.InputTokens+resp.Usage.OutputTokens)
		})
	}
}

func TestAuditAnthropicAgentBudget(t *testing.T) {
	for _, normalized := range []bool{false, true} {
		name := "current"
		if normalized {
			name = "test_only_normalized_input"
		}
		t.Run(name, func(t *testing.T) {
			client, transport := auditClient(t, "anthropic", auditAnthropicFirst, auditAnthropicSecond)
			if normalized {
				client = auditNormalizedClient{client}
			}
			wf := dag.WorkflowView{Spec: dag.GraphSpec{Budget: dag.Budget{MaxTokens: 18}}}
			executor := newTestExecutor(t, ExecutorDeps{Snapshot: func(context.Context, string) (dag.WorkflowView, error) { return wf, nil }})
			hook := executor.budgetHook("audit")
			var seen []llm.Usage
			result, err := agent.Run(context.Background(), agent.RunSpec{
				Client: client, Prompt: "audit", MaxTurns: 2,
				Submit: agent.SubmitSpec{Decode: func(raw json.RawMessage) (any, error) { return string(raw), nil }},
				BeforeTurn: func(ctx context.Context, usage llm.Usage) error {
					seen = append(seen, usage)
					return hook(ctx, usage)
				},
			})
			if len(seen) != 2 || result == nil {
				t.Fatalf("seen=%+v result=%+v err=%v", seen, result, err)
			}
			if normalized {
				if !errors.Is(err, errTokenBudgetExhausted) || transport.calls.Load() != 1 || seen[1].InputTokens+seen[1].OutputTokens != 20 || result.Turns != 1 {
					t.Fatalf("err=%v requests=%d seen=%+v result=%+v", err, transport.calls.Load(), seen, result)
				}
			} else if err != nil || transport.calls.Load() != 2 || seen[1].InputTokens+seen[1].OutputTokens != 15 || result.Turns != 2 {
				t.Fatalf("err=%v requests=%d seen=%+v result=%+v", err, transport.calls.Load(), seen, result)
			}
			t.Logf("limit=18 synthetic first usage=%+v checked=%d requests=%d turns=%d err=%v", seen[1], seen[1].InputTokens+seen[1].OutputTokens, transport.calls.Load(), result.Turns, err)
		})
	}
}

func TestAuditAnthropicDAGSchedulerBudget(t *testing.T) {
	for _, normalized := range []bool{false, true} {
		name := "current"
		if normalized {
			name = "test_only_normalized_input"
		}
		t.Run(name, func(t *testing.T) {
			client, transport := auditClient(t, "anthropic", auditAnthropicFirst, auditAnthropicSecond)
			if normalized {
				client = auditNormalizedClient{client}
			}
			executor := newTestExecutor(t, ExecutorDeps{})
			engine := dag.NewEngine(dag.NewMemoryEventStore(), auditDAGExecutor{client: client, review: executor}, nil, dag.Options{MaxConcurrentNodes: 1})
			workflowID, err := engine.Create(context.Background(), dag.GraphSpec{
				Nodes: []dag.NodeSpec{
					{ID: "a", Kind: "audit", Required: true},
					{ID: "b", Kind: "audit", Required: true, DependsOn: []string{"a"}},
				}, Budget: dag.Budget{MaxTokens: 18},
			}, nil)
			if err != nil {
				t.Fatal(err)
			}
			if err := engine.Run(context.Background(), workflowID); err != nil {
				t.Fatal(err)
			}
			wf, err := engine.Snapshot(context.Background(), workflowID)
			if err != nil {
				t.Fatal(err)
			}
			a, b := wf.Nodes["a"], wf.Nodes["b"]
			if normalized {
				if a.Usage.InputTokens+a.Usage.OutputTokens != 20 || transport.calls.Load() != 1 || b.Status != dag.NodeFailed || b.Reason != "budget_exhausted" {
					t.Fatalf("a=%+v b=%+v requests=%d", a, b, transport.calls.Load())
				}
			} else if a.Usage.InputTokens+a.Usage.OutputTokens != 15 || transport.calls.Load() != 2 || b.Status != dag.NodeCompleted {
				t.Fatalf("a=%+v b=%+v requests=%d", a, b, transport.calls.Load())
			}
			t.Logf("limit=18 first node usage=%+v checked=%d second_node=%s reason=%s requests=%d", a.Usage, a.Usage.InputTokens+a.Usage.OutputTokens, b.Status, b.Reason, transport.calls.Load())
		})
	}
}
